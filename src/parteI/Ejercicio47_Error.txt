// EJEMPLO ERRADO

public class Perro {
    private String nombre;

    public Perro(String nombre) {
        this.nombre = nombre;
    }

    public void ladrar() {
        System.out.println(this.nombre + " dice: ¡Guau!");
    }
}

public class Ejercicio47Error {
    public static void main(String[] args) {
        Perro perro; // se declara la referencia, pero nunca se crea el objeto
        perro.ladrar(); // se usa sin haber hecho "new Perro(...)"
    }
}
